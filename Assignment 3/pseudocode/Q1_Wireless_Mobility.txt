ALGORITHM WirelessMobilitySimulation(Nodes, AccessPoints, Packets, Scenario)
INPUT: Mobile nodes, access points, packet count, mobility scenario
OUTPUT: Delivery count, loss count, average delay, final attachment
BEGIN
  SELECT initial access point for each mobile node
  FOR each packet DO
    DETECT current signal and link state
    IF current access point is reachable THEN
      ADD base transmission delay
      DELIVER packet if transmission succeeds
    ELSE
      SEARCH for reachable neighbouring access point
      IF candidate signal >= MIN_SIGNAL THEN
        SWITCH attachment to candidate access point
        RECORD handoff delay
        RESUME transmission
      ELSE
        DROP packet and record interruption
      END IF
    END IF
  END FOR
  CALCULATE PDR and average delay
  RETURN metrics
END
